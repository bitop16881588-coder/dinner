async function submitReg() { const name = document.getElementById('reg-name').value; if (!name) return alert("請輸入姓名"); // 將資料打包 const payload = { type: 'reg', name: name, rest: '陳記火鍋', date: '2026-06-05' }; try { const response = await fetch(SCRIPT_URL, { method: 'POST', mode: 'no-cors', // 這是解決跨域錯誤最重要的一行 headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }); alert("報名成功！"); location.reload(); } catch (error) { alert("發送失敗，請確認網路連線"); console.error("Error:", error); } }
